    public function startRow(): int

        return 2;
    }

    public function generateRunningCode()
    {
        $number = mt_rand(1000000000000, 9999999999999);
        if ($this->runningCodeExists($number)) {
            return generateRunningCode();
        }
        return $number;
    }

    public function runningCodeExists($number) {
        return ProductStock::where(['ps_running_code' => $number])->exists();
    }

    public function collection(Collection $row)
    {
        foreach ($row as $r) {
            if ($r[0] == null) {
                return null;
            }

            $p_id = DB::table('products')->insertGetId([
                'br_id' => $this->br_id,
                'pc_id' => $this->pc_id,
                'psc_id' => $this->psc_id,
                'pssc_id' => $this->pssc_id,
                'mc_id' => $this->mc_id,
                'ps_id' => $this->ps_id,
                'pu_id' => $this->pu_id,
                'gn_id' => $this->gn_id,
                'ss_id' => $this->ss_id,
                'p_name' => $r[0],
                'p_aging' => $r[1],
                'p_color' => $r[2],
                'p_price_tag' => $r[3],
                'p_purchase_price' => $r[4],
                'p_sell_price' => $r[5],
                'p_delete' => '0',
            ]);

            $exp = explode(',', $r[6]);
            $count = (Integer)count($exp);
            for ($i=0; $i<=$count; $i++) {
                if (empty($exp[$i])) {
                    continue;
                }
                $exp_ps = explode('-', $exp[$i]);
                $size = Size::where(['sz_name' => $exp_ps[0]]);
                if (!empty($size->first()->id)) {
                    $sz_id = $size->first()->id;
                } else {
                    return false;
                }
                ProductStock::create([
                    'p_id' => $p_id,
                    'sz_id' => $sz_id,
                    'ps_qty' => '0',
                    'ps_barcode' => $exp_ps[1],
                    'ps_running_code' => $this->generateRunningCode()
                ]);
            }
        }
    }
